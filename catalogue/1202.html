Dephased $12\times 12$ orbits can be obtained, using the Diţă's method, from $F_2$ and dephased $6\times 6$ complex Hadamard matrices: $F_6^{(2)}$, $C_6^{(0)}$ and $S_6^{(0)}$. They are: $$\begin{aligned} FD_{12}^{(8)}(a,b,c,d,e,f,g,h) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot F_6^{(2)}(a,b) & \big[F_2\big]_{1,2}\cdot \mathbb{D}(d,e,f,g,h)\cdot D_6^{(1)}(c)\\ \hline \big[F_2\big]_{2,1}\cdot F_6^{(2)}(a,b) & \big[F_2\big]_{2,2}\cdot \mathbb{D}(d,e,f,g,h)\cdot D_6^{(1)}(c)\\ \end{array}\right], \\ \\ \\ FC_{12}^{(7)}(a,b,c,d,e,f,g) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot F_6^{(2)}(a,b) & \big[F_2\big]_{1,2}\cdot \mathbb{D}(c,d,e,f,g)\cdot C_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot F_6^{(2)}(a,b) & \big[F_2\big]_{2,2}\cdot \mathbb{D}(c,d,e,f,g)\cdot C_6^{(0)}\\ \end{array}\right], \\ \\ \\ FS_{12}^{(7)}(a,b,c,d,e,f,g) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot F_6^{(2)}(a,b) & \big[F_2\big]_{1,2}\cdot \mathbb{D}(c,d,e,f,g)\cdot S_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot F_6^{(2)}(a,b) & \big[F_2\big]_{2,2}\cdot \mathbb{D}(c,d,e,f,g)\cdot S_6^{(0)}\\ \end{array}\right], \\ \\ \\ DD_{12}^{(7)}(a,b,c,d,e,f,g) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot D_6^{(1)}(a) & \big[F_2\big]_{1,2}\cdot \mathbb{D}(c,d,e,f,g)\cdot D_6^{(1)}(b)\\ \hline \big[F_2\big]_{2,1}\cdot D_6^{(1)}(a) & \big[F_2\big]_{2,2}\cdot \mathbb{D}(c,d,e,f,g)\cdot D_6^{(1)}(b)\\ \end{array}\right], \\ \\ \\ DC_{12}^{(6)}(a,b,c,d,e,f) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot D_6^{(1)}(a) & \big[F_2\big]_{1,2}\cdot \mathbb{D}(b,c,d,e,f)\cdot C_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot D_6^{(1)}(a) & \big[F_2\big]_{2,2}\cdot \mathbb{D}(b,c,d,e,f)\cdot C_6^{(0)}\\ \end{array}\right], \\ \\ \\ DS_{12}^{(6)}(a,b,c,d,e,f) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot D_6^{(1)}(a) & \big[F_2\big]_{1,2}\cdot \mathbb{D}(b,c,d,e,f)\cdot S_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot D_6^{(1)}(a) & \big[F_2\big]_{2,2}\cdot \mathbb{D}(b,c,d,e,f)\cdot S_6^{(0)}\\ \end{array}\right], \\ \\ \\ CC_{12}^{(5)}(a,b,c,d,e) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot C_6^{(0)} & \big[F_2\big]_{1,2}\cdot \mathbb{D}(a,b,c,d,e)\cdot C_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot C_6^{(0)} & \big[F_2\big]_{2,2}\cdot \mathbb{D}(a,b,c,d,e)\cdot C_6^{(0)}\\ \end{array}\right], \\ \\ \\ CS_{12}^{(5)}(a,b,c,d,e) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot C_6^{(0)} & \big[F_2\big]_{1,2}\cdot \mathbb{D}(a,b,c,d,e)\cdot S_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot C_6^{(0)} & \big[F_2\big]_{2,2}\cdot \mathbb{D}(a,b,c,d,e)\cdot S_6^{(0)}\\ \end{array}\right], \\ \\ \\ SS_{12}^{(5)}(a,b,c,d,e) &= \left[\begin{array}{l|l} \big[F_2\big]_{1,1}\cdot S_6^{(0)} & \big[F_2\big]_{1,2}\cdot \mathbb{D}(a,b,c,d,e)\cdot S_6^{(0)}\\ \hline \big[F_2\big]_{2,1}\cdot S_6^{(0)} &
